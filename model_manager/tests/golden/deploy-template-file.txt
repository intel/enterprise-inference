apiVersion: batch/v1
kind: Job
metadata:
  name: download-tmpl-file
  namespace: ns1
  labels:
    app.kubernetes.io/name: download-tmpl-file
    app.kubernetes.io/managed-by: model-manager
spec:
  backoffLimit: 2
  activeDeadlineSeconds: 7200
  # Garbage-collect the finished Job (and its pod) automatically once it
  # completes or fails, so a successful download never leaves a stray pod
  # behind even if the model-manager CLI has already exited. The window is
  # long enough to inspect logs of a failed job before it is reaped.
  ttlSecondsAfterFinished: 300
  template:
    metadata:
      labels:
        app.kubernetes.io/name: download-tmpl-file
        app.kubernetes.io/managed-by: model-manager
    spec:
      restartPolicy: Never
      securityContext:
        runAsNonRoot: true
        # uid/gid match the serving pod for this runtime so prepared files are
        # readable by the server (vLLM=1000, OVMS=5000). fsGroup makes the kubelet
        # chown the mounted volume to this GID and mount it group-writable —
        # without it, dynamically provisioned volumes (e.g. rancher.io/local-path)
        # are created root:root 0755 and the non-root job gets "Permission denied".
        runAsUser: 1000
        runAsGroup: 1000
        fsGroup: 1000
        seccompProfile:
          type: RuntimeDefault
      containers:
        - name: downloader
          image: img/vllm:0.24.0
          command:
            - sh
            - -c
            - |
              set -e
              export HOME=/tmp HF_HOME=/tmp/.hf XDG_CACHE_HOME=/tmp/.cache PIP_CACHE_DIR=/tmp/.pip
              if [ -f /models/tmpl-file/.download_complete ]; then echo 'Weights already present, skipping download'; exit 0; fi
              command -v hf >/dev/null 2>&1 || pip install -q huggingface_hub[hf_xet]
              hf download 'org/file' --local-dir /models/tmpl-file
              touch /models/tmpl-file/.download_complete
          securityContext:
            allowPrivilegeEscalation: false
            capabilities:
              drop:
              - ALL
            seccompProfile:
              type: RuntimeDefault
          # Bound the job (CWE-400). Requests stay small so this transient job
          # schedules easily; the memory limit tracks the model's serving
          # envelope so OVMS `ovms --pull` conversion/quantization does not OOM.
          resources:
            requests:
              cpu: "500m"
              memory: "2Gi"
            limits:
              cpu: "4"
              memory: "8Gi"
          env:
            - name: HOME
              value: "/models"
            - name: HF_HOME
              value: "/models/.hf-cache"
          volumeMounts:
            - name: model-storage
              mountPath: /models
      volumes:
        - name: model-storage
          persistentVolumeClaim:
            claimName: model-store
